Apps Script Code (Extensions → Apps Script තුළට Paste කරන්න): 
const SHEET_ID = '1fzn7kAJQNFtG3TqhMfPRsQEXV8cqUP7HnfyAUzz01SY';
const HEADERS = ['Date', 'Contact', 'Name', 'ISP Package name', 'GB', 'price', 'Configuration', 'Subscribtion usage(link)'];

// STEP 2: Select this function in Apps Script toolbar and click "Run" once to Authorize & Format your Sheet!
function setupSheetAndAuthorize() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = getOrCreateMonthSheet(ss, 'October');
  renderMonthSheetData(sheet, []);
}

function getOrCreateMonthSheet(ss, monthName) {
  const cleanName = String(monthName || 'October').trim();
  let sheet = ss.getSheetByName(cleanName);
  if (!sheet) {
    // Also check if a tab has trailing spaces like "October "
    const allSheets = ss.getSheets();
    for (let i = 0; i < allSheets.length; i++) {
      if (allSheets[i].getName().trim().toLowerCase() === cleanName.toLowerCase()) {
        sheet = allSheets[i];
        break;
      }
    }
  }
  if (!sheet) {
    sheet = ss.insertSheet(cleanName);
  }
  return sheet;
}

function renderMonthSheetData(sheet, records) {
  sheet.clearContents();
  sheet.clearFormats();
  sheet.appendRow(HEADERS);
  sheet.getRange(1, 1, 1, HEADERS.length)
       .setBackground('#999999')
       .setFontColor('#000000')
       .setFontWeight('bold')
       .setHorizontalAlignment('center');

  let totalProfit = 0;
  let totalRemainingGB = 0;

  const rows = (records || []).map(r => {
    const price = parseFloat(r.price) || 0;
    totalProfit += price;
    if (String(r.gb).toUpperCase() !== 'UNLIMITED') {
      const quota = parseFloat(r.gb) || 0;
      const used = parseFloat(r.usedGb) || 0;
      totalRemainingGB += Math.max(0, quota - used);
    }
    return [
      r.date || '',
      r.contact || '',
      r.name || '',
      r.ispPackage || '',
      r.gb || '',
      price,
      r.configuration || '',
      r.subLink || ''
    ];
  });

  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, HEADERS.length).setValues(rows);
  }

  // Row 21 (or 3 rows below data): Column E = remaining data, Column F = bright red total profit
  const summaryRowIndex = Math.max(rows.length + 4, 21);
  sheet.getRange(summaryRowIndex, 5)
       .setValue('remaining data  ' + totalRemainingGB.toFixed(1) + ' GB')
       .setFontWeight('bold');
  sheet.getRange(summaryRowIndex, 6)
       .setValue('total profit  ' + totalProfit + ' LKR')
       .setBackground('#ff0000')
       .setFontColor('#000000')
       .setFontWeight('bold');
  SpreadsheetApp.flush();
}

function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const monthName = (e && e.parameter && e.parameter.month) ? e.parameter.month : 'October';
    const sheet = getOrCreateMonthSheet(ss, monthName);
    const values = sheet.getDataRange().getValues();
    return ContentService.createTextOutput(JSON.stringify({
      status: 'ok',
      sheetName: sheet.getName(),
      rows: values
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    const payload = JSON.parse(raw);

    if (payload.action === 'SYNC_MONTH' && payload.monthName && Array.isArray(payload.records)) {
      const sheet = getOrCreateMonthSheet(ss, payload.monthName);
      renderMonthSheetData(sheet, payload.records);
    } else if (payload.action === 'SYNC_ALL_MONTHS' && payload.groupedByMonth) {
      Object.keys(payload.groupedByMonth).forEach(monthName => {
        const sheet = getOrCreateMonthSheet(ss, monthName);
        renderMonthSheetData(sheet, payload.groupedByMonth[monthName]);
      });
    }
    return ContentService.createTextOutput(JSON.stringify({
      status: 'ok',
      syncedAt: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
