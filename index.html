function getOrCreateMonthSheet(ss, monthName) {
  const cleanName = String(monthName || 'October').trim();
  let sheet = ss.getSheetByName(cleanName);
  if (!sheet) {
    const allSheets = ss.getSheets();
    for (let i = 0; i < allSheets.length; i++) {
      if (allSheets[i].getName().trim().toLowerCase() === cleanName.toLowerCase()) {
        sheet = allSheets[i];
        break;
      }
    }
  }
  if (!sheet) sheet = ss.insertSheet(cleanName);
  return sheet;
}

function renderMonthSheetData(sheet, records, allowEmptyOverwrite) {
  const safeRecords = Array.isArray(records) ? records : [];
  if (safeRecords.length === 0 && !allowEmptyOverwrite && sheet.getLastRow() > 1) return;

  sheet.clearContents();
  sheet.clearFormats();
  sheet.appendRow(HEADERS);
  sheet.getRange(1, 1, 1, HEADERS.length).setBackground('#999999').setFontColor('#000000').setFontWeight('bold').setHorizontalAlignment('center');

  let totalProfit = 0;
  let totalRemainingGB = 0;
  const rows = safeRecords.map(r => {
    const price = parseFloat(r.price) || 0;
    totalProfit += price;
    if (String(r.gb).toUpperCase() !== 'UNLIMITED') {
      const quota = parseFloat(r.gb) || 0;
      const used = parseFloat(r.usedGb) || 0;
      totalRemainingGB += Math.max(0, quota - used);
    }
    return [r.date || '', r.contact || '', r.name || '', r.ispPackage || '', r.gb || '', price, r.configuration || '', r.subLink || '', r.billingId || '', r.adminRemark || r.enteredBy || 'CEO Thenul'];
  });

  if (rows.length > 0) sheet.getRange(2, 1, rows.length, HEADERS.length).setValues(rows);
  const summaryRowIndex = Math.max(rows.length + 4, 21);
  sheet.getRange(summaryRowIndex, 5).setValue('remaining data  ' + totalRemainingGB.toFixed(1) + ' GB').setFontWeight('bold');
  sheet.getRange(summaryRowIndex, 6).setValue('total profit  ' + totalProfit + ' LKR').setBackground('#ff0000').setFontColor('#000000').setFontWeight('bold');
  SpreadsheetApp.flush();
}
